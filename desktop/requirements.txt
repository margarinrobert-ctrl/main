# Runtime dependencies for Trading Backtester.
#
# Deliberately small: the application does its own charting on top of pyqtgraph
# and its own reporting on top of Qt, so there is no matplotlib, no reportlab and
# no web stack to ship, install or keep patched.

PySide6>=6.6,<7          # Qt 6 bindings (LGPL) -- UI toolkit
pyqtgraph>=0.13          # fast plotting on top of Qt's scene graph
numpy>=1.26              # array maths for indicators and the engine
pandas>=2.0              # CSV parsing, timezone handling, period grouping
pyarrow>=14.0            # parquet storage for imported datasets (optional at runtime)
